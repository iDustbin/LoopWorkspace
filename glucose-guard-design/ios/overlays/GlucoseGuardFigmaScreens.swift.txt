private enum GlucoseGuardDefaults {
    static let low = "gg.lowMgdl"
    static let high = "gg.highMgdl"
    static let normal = "gg.normalNotification"
    static let alarm = "gg.alarmNotification"
    static let live = "gg.liveActivities"
    static let reading = "gg.glucoseReading"
    static let interval = "gg.updateInterval"
    static let lowAlarm = "gg.lowAlarm"
    static let highAlarm = "gg.highAlarm"
    static let connectionAlarm = "gg.connectionAlarm"
    static let volume = "gg.alarmVolume"
    static let ignoreMute = "gg.ignoreMute"
    static let phoneOn = "gg.phoneEnabled"
    static let phone = "gg.phoneNumber"
    static let whatsOn = "gg.whatsAppEnabled"
    static let whats = "gg.whatsAppNumber"
    static let libreNick = "gg.libreNickname"
    static let libreServer = "gg.libreServer"
    static let dexNick = "gg.dexcomNickname"
    static let dexServer = "gg.dexcomServer"
    static let lastAlert = "gg.lastAlertAt"
}

final class GlucoseGuardPreferences: ObservableObject {
    static let shared = GlucoseGuardPreferences()
    private let defaults = UserDefaults.standard

    @Published var lowMgdl: Double { didSet { defaults.set(lowMgdl, forKey: GlucoseGuardDefaults.low) } }
    @Published var highMgdl: Double { didSet { defaults.set(highMgdl, forKey: GlucoseGuardDefaults.high) } }
    @Published var normalNotification: Bool { didSet { defaults.set(normalNotification, forKey: GlucoseGuardDefaults.normal) } }
    @Published var alarmNotification: Bool { didSet { defaults.set(alarmNotification, forKey: GlucoseGuardDefaults.alarm) } }
    @Published var liveActivities: Bool { didSet { defaults.set(liveActivities, forKey: GlucoseGuardDefaults.live) } }
    @Published var glucoseReading: Bool { didSet { defaults.set(glucoseReading, forKey: GlucoseGuardDefaults.reading) } }
    @Published var updateIntervalMinutes: Int { didSet { defaults.set(updateIntervalMinutes, forKey: GlucoseGuardDefaults.interval) } }
    @Published var lowAlarm: String { didSet { defaults.set(lowAlarm, forKey: GlucoseGuardDefaults.lowAlarm) } }
    @Published var highAlarm: String { didSet { defaults.set(highAlarm, forKey: GlucoseGuardDefaults.highAlarm) } }
    @Published var connectionAlarm: String { didSet { defaults.set(connectionAlarm, forKey: GlucoseGuardDefaults.connectionAlarm) } }
    @Published var alarmVolume: Double { didSet { defaults.set(alarmVolume, forKey: GlucoseGuardDefaults.volume) } }
    @Published var ignoreMute: Bool { didSet { defaults.set(ignoreMute, forKey: GlucoseGuardDefaults.ignoreMute) } }
    @Published var phoneEnabled: Bool { didSet { defaults.set(phoneEnabled, forKey: GlucoseGuardDefaults.phoneOn) } }
    @Published var phoneNumber: String { didSet { defaults.set(phoneNumber, forKey: GlucoseGuardDefaults.phone) } }
    @Published var whatsAppEnabled: Bool { didSet { defaults.set(whatsAppEnabled, forKey: GlucoseGuardDefaults.whatsOn) } }
    @Published var whatsAppNumber: String { didSet { defaults.set(whatsAppNumber, forKey: GlucoseGuardDefaults.whats) } }
    @Published var libreNickname: String { didSet { defaults.set(libreNickname, forKey: GlucoseGuardDefaults.libreNick) } }
    @Published var libreServer: String { didSet { defaults.set(libreServer, forKey: GlucoseGuardDefaults.libreServer) } }
    @Published var dexcomNickname: String { didSet { defaults.set(dexcomNickname, forKey: GlucoseGuardDefaults.dexNick) } }
    @Published var dexcomServer: String { didSet { defaults.set(dexcomServer, forKey: GlucoseGuardDefaults.dexServer) } }

    init() {
        let store = UserDefaults.standard
        lowMgdl = store.object(forKey: GlucoseGuardDefaults.low) as? Double ?? 90
        highMgdl = store.object(forKey: GlucoseGuardDefaults.high) as? Double ?? 187
        normalNotification = store.object(forKey: GlucoseGuardDefaults.normal) as? Bool ?? true
        alarmNotification = store.object(forKey: GlucoseGuardDefaults.alarm) as? Bool ?? true
        liveActivities = store.object(forKey: GlucoseGuardDefaults.live) as? Bool ?? false
        glucoseReading = store.object(forKey: GlucoseGuardDefaults.reading) as? Bool ?? true
        updateIntervalMinutes = store.object(forKey: GlucoseGuardDefaults.interval) as? Int ?? 1
        lowAlarm = store.string(forKey: GlucoseGuardDefaults.lowAlarm) ?? "vibration"
        highAlarm = store.string(forKey: GlucoseGuardDefaults.highAlarm) ?? "none"
        connectionAlarm = store.string(forKey: GlucoseGuardDefaults.connectionAlarm) ?? "vibration"
        alarmVolume = store.object(forKey: GlucoseGuardDefaults.volume) as? Double ?? 0
        ignoreMute = store.bool(forKey: GlucoseGuardDefaults.ignoreMute)
        phoneEnabled = store.bool(forKey: GlucoseGuardDefaults.phoneOn)
        phoneNumber = store.string(forKey: GlucoseGuardDefaults.phone) ?? ""
        whatsAppEnabled = store.bool(forKey: GlucoseGuardDefaults.whatsOn)
        whatsAppNumber = store.string(forKey: GlucoseGuardDefaults.whats) ?? ""
        libreNickname = store.string(forKey: GlucoseGuardDefaults.libreNick) ?? ""
        libreServer = store.string(forKey: GlucoseGuardDefaults.libreServer) ?? "APAC"
        dexcomNickname = store.string(forKey: GlucoseGuardDefaults.dexNick) ?? ""
        dexcomServer = store.string(forKey: GlucoseGuardDefaults.dexServer) ?? "APAC"
    }

    var lastAlertAt: Date? {
        get { defaults.object(forKey: GlucoseGuardDefaults.lastAlert) as? Date }
        set { defaults.set(newValue, forKey: GlucoseGuardDefaults.lastAlert) }
    }
}

final class GlucoseGuardAlertCenter: NSObject, UNUserNotificationCenterDelegate {
    static let shared = GlucoseGuardAlertCenter()

    func install() {
        let center = UNUserNotificationCenter.current()
        center.delegate = self
        let call = UNNotificationAction(identifier: "GG_CALL", title: NSLocalizedString("Call", comment: "Call emergency contact"), options: [.foreground])
        let whats = UNNotificationAction(identifier: "GG_WHATSAPP", title: "WhatsApp", options: [.foreground])
        let category = UNNotificationCategory(identifier: "GG_GLUCOSE", actions: [call, whats], intentIdentifiers: [])
        center.setNotificationCategories([category])
        center.requestAuthorization(options: [.alert, .sound, .badge]) { _, _ in }
    }

    func userNotificationCenter(_ center: UNUserNotificationCenter, didReceive response: UNNotificationResponse, withCompletionHandler completionHandler: @escaping () -> Void) {
        let prefs = GlucoseGuardPreferences.shared
        if response.actionIdentifier == "GG_CALL" {
            GlucoseGuardContactActions.call(prefs.phoneNumber)
        } else if response.actionIdentifier == "GG_WHATSAPP" {
            GlucoseGuardContactActions.whatsApp(prefs.whatsAppNumber)
        }
        completionHandler()
    }

    func evaluate(mgdl: Double?, hasCGM: Bool) {
        let prefs = GlucoseGuardPreferences.shared
        guard prefs.alarmNotification || prefs.normalNotification else { return }
        if let last = prefs.lastAlertAt, Date().timeIntervalSince(last) < 15 * 60 {
            return
        }
        if let mgdl {
            if mgdl <= prefs.lowMgdl, prefs.lowAlarm != "none", prefs.alarmNotification {
                post("Low Glucose", body: String(format: "%.0f mg/dL", mgdl), sound: prefs.lowAlarm == "vibration" ? false : true)
                prefs.lastAlertAt = Date()
                return
            }
            if mgdl >= prefs.highMgdl, prefs.highAlarm != "none", prefs.alarmNotification {
                post("High Glucose", body: String(format: "%.0f mg/dL", mgdl), sound: prefs.highAlarm == "vibration" ? false : true)
                prefs.lastAlertAt = Date()
                return
            }
        } else if !hasCGM, prefs.connectionAlarm != "none" {
            post("Connection Alarm", body: NSLocalizedString("No CGM connection", comment: "Missing CGM alarm"), sound: false)
            prefs.lastAlertAt = Date()
        }
    }

    private func post(_ title: String, body: String, sound: Bool) {
        let prefs = GlucoseGuardPreferences.shared
        let content = UNMutableNotificationContent()
        content.title = title
        content.body = body
        content.categoryIdentifier = "GG_GLUCOSE"
        if sound && prefs.alarmVolume > 0 {
            content.sound = .default
        }
        if prefs.ignoreMute || prefs.liveActivities {
            content.interruptionLevel = .timeSensitive
        }
        let request = UNNotificationRequest(identifier: "gg-\(title)", content: content, trigger: nil)
        UNUserNotificationCenter.current().add(request, withCompletionHandler: nil)
    }
}

private enum GlucoseGuardContactActions {
    static func digits(_ raw: String) -> String {
        raw.filter { $0.isNumber }
    }

    static func call(_ raw: String) {
        let number = digits(raw)
        guard !number.isEmpty, let url = URL(string: "tel://\(number)") else { return }
        UIApplication.shared.open(url)
    }

    static func whatsApp(_ raw: String) {
        let number = digits(raw)
        guard !number.isEmpty, let url = URL(string: "https://wa.me/\(number)") else { return }
        UIApplication.shared.open(url)
    }
}

struct GlucoseGuardGlucoseSettingsView: View {
    let unit: HKUnit
    let transmitter: String
    let onConnections: () -> Void
    let onEmergency: () -> Void
    @ObservedObject private var prefs = GlucoseGuardPreferences.shared

    var body: some View {
        GlucoseGuardScreenChrome(NSLocalizedString("Glucose Settings", comment: "Glucose settings title")) {
            VStack(alignment: .leading, spacing: 16) {
                GlucoseGuardGlass {
                    VStack(alignment: .leading, spacing: 16) {
                        HStack {
                            Text(NSLocalizedString("Unit", comment: "Glucose unit"))
                                .foregroundColor(.white)
                            Spacer()
                            Text(unit.unitString).foregroundColor(GlucoseGuardTheme.red)
                        }
                        slider(
                            NSLocalizedString("Low", comment: "Low glucose slider"),
                            value: $prefs.lowMgdl,
                            range: 54...120
                        )
                        slider(
                            NSLocalizedString("High", comment: "High glucose slider"),
                            value: $prefs.highMgdl,
                            range: 140...250
                        )
                    }
                }

                GlucoseGuardGlass {
                    VStack(spacing: 0) {
                        GlucoseGuardToggleRow(title: NSLocalizedString("Normal Glucose Notification", comment: "Normal glucose toggle"), isOn: $prefs.normalNotification)
                        GlucoseGuardToggleRow(title: NSLocalizedString("Alarm Glucose Notification", comment: "Alarm glucose toggle"), isOn: $prefs.alarmNotification)
                        GlucoseGuardToggleRow(title: NSLocalizedString("Glucose Live Activities", comment: "Live Activities toggle"), isOn: $prefs.liveActivities)
                        GlucoseGuardToggleRow(title: NSLocalizedString("Glucose Reading", comment: "Show glucose reading toggle"), isOn: $prefs.glucoseReading)
                    }
                }

                GlucoseGuardGlass {
                    VStack(alignment: .leading, spacing: 10) {
                        Text(NSLocalizedString("Sensor Connection", comment: "Sensor connection section"))
                            .font(.headline)
                            .foregroundColor(.white)
                        labeled(NSLocalizedString("Transmitter", comment: "Transmitter label"), transmitter)
                        labeled(NSLocalizedString("Update Interval", comment: "CGM interval"), intervalText)
                        GlucoseGuardNavRow(
                            title: NSLocalizedString("Connection settings", comment: "Open connection settings"),
                            action: onConnections
                        )
                    }
                }

                GlucoseGuardGlass {
                    VStack(alignment: .leading, spacing: 10) {
                        Text(NSLocalizedString("Alarm Settings", comment: "Alarm settings section"))
                            .font(.headline)
                            .foregroundColor(.white)
                        alarmRow(NSLocalizedString("Low Glucose Alarm", comment: "Low alarm"), value: $prefs.lowAlarm)
                        alarmRow(NSLocalizedString("High Glucose Alarm", comment: "High alarm"), value: $prefs.highAlarm)
                        alarmRow(NSLocalizedString("Connection Alarm", comment: "Connection alarm"), value: $prefs.connectionAlarm)
                        VStack(alignment: .leading, spacing: 6) {
                            HStack {
                                Text(NSLocalizedString("Alarm Volume", comment: "Alarm volume"))
                                    .foregroundColor(.white)
                                Spacer()
                                Text("\(Int(prefs.alarmVolume))%")
                                    .foregroundColor(GlucoseGuardTheme.muted)
                            }
                            Slider(value: $prefs.alarmVolume, in: 0...100)
                                .tint(GlucoseGuardTheme.red)
                        }
                        GlucoseGuardToggleRow(title: NSLocalizedString("Ignore Mute", comment: "Ignore mute toggle"), isOn: $prefs.ignoreMute)
                        GlucoseGuardNavRow(
                            title: NSLocalizedString("Emergency Contacts", comment: "Open emergency contacts"),
                            action: onEmergency
                        )
                    }
                }
            }
        }
    }

    private var intervalText: String {
        prefs.updateIntervalMinutes <= 1
            ? NSLocalizedString("Jede Minute", comment: "Update every minute")
            : String(format: NSLocalizedString("Every %d min", comment: "Update interval minutes"), prefs.updateIntervalMinutes)
    }

    private func labeled(_ title: String, _ value: String) -> some View {
        HStack {
            Text(title).foregroundColor(GlucoseGuardTheme.muted)
            Spacer()
            Text(value).foregroundColor(GlucoseGuardTheme.red)
        }
        .font(.subheadline)
    }

    private func slider(_ title: String, value: Binding<Double>, range: ClosedRange<Double>) -> some View {
        let display = HKQuantity(unit: .milligramsPerDeciliter, doubleValue: value.wrappedValue).doubleValue(for: unit)
        return VStack(alignment: .leading, spacing: 6) {
            HStack {
                Text(title).foregroundColor(.white)
                Spacer()
                Text(String(format: unit.unitString.contains("mol") ? "%.1f" : "%.0f", display))
                    .foregroundColor(GlucoseGuardTheme.red)
            }
            Slider(value: value, in: range)
                .tint(GlucoseGuardTheme.red)
        }
    }

    private func alarmRow(_ title: String, value: Binding<String>) -> some View {
        HStack {
            Text(title).foregroundColor(.white)
            Spacer()
            Menu {
                Button(NSLocalizedString("Vibration", comment: "Alarm vibration")) { value.wrappedValue = "vibration" }
                Button(NSLocalizedString("None", comment: "Alarm off")) { value.wrappedValue = "none" }
            } label: {
                Text(value.wrappedValue == "none"
                     ? NSLocalizedString("None", comment: "Alarm off")
                     : NSLocalizedString("Vibration", comment: "Alarm vibration"))
                    .foregroundColor(GlucoseGuardTheme.red)
            }
        }
        .padding(.vertical, 6)
    }
}

struct GlucoseGuardEmergencyContactsView: View {
    @ObservedObject private var prefs = GlucoseGuardPreferences.shared

    var body: some View {
        GlucoseGuardScreenChrome(NSLocalizedString("Emergency Contacts", comment: "Emergency contacts title")) {
            VStack(alignment: .leading, spacing: 16) {
                contactCard(
                    title: NSLocalizedString("Phone Number", comment: "Emergency phone section"),
                    enabled: $prefs.phoneEnabled,
                    number: $prefs.phoneNumber,
                    actionTitle: NSLocalizedString("Call", comment: "Call emergency contact")
                ) {
                    GlucoseGuardContactActions.call(prefs.phoneNumber)
                }
                contactCard(
                    title: "WhatsApp",
                    enabled: $prefs.whatsAppEnabled,
                    number: $prefs.whatsAppNumber,
                    actionTitle: "WhatsApp"
                ) {
                    GlucoseGuardContactActions.whatsApp(prefs.whatsAppNumber)
                }
            }
        }
    }

    private func contactCard(title: String, enabled: Binding<Bool>, number: Binding<String>, actionTitle: String, action: @escaping () -> Void) -> some View {
        GlucoseGuardGlass {
            VStack(alignment: .leading, spacing: 12) {
                GlucoseGuardToggleRow(title: title, isOn: enabled)
                Text(NSLocalizedString("Used when a glucose or connection alarm fires.", comment: "Emergency contact help"))
                    .font(.footnote)
                    .foregroundColor(GlucoseGuardTheme.muted)
                TextField(title, text: number)
                    .keyboardType(.phonePad)
                    .foregroundColor(.white)
                    .padding(12)
                    .background(Color.white.opacity(0.06))
                    .clipShape(RoundedRectangle(cornerRadius: 12, style: .continuous))
                if enabled.wrappedValue {
                    Button(action: action) {
                        Text(actionTitle)
                            .font(.headline)
                            .foregroundColor(GlucoseGuardTheme.red)
                            .frame(maxWidth: .infinity)
                    }
                    .buttonStyle(.plain)
                }
            }
        }
    }
}
