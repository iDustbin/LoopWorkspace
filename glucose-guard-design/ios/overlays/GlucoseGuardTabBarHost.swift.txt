    // GLUCOSE_GUARD_XCODE_DESIGN host

    private let glucoseGuardHomeState = GlucoseGuardHomeState()
    private var glucoseGuardTabBarHost: UIHostingController<GlucoseGuardTabBar>?
    private var glucoseGuardHomeHUDHost: UIHostingController<GlucoseGuardHomeHUD>?

    private func installGlucoseGuardTabBar() {
        overrideUserInterfaceStyle = .dark
        tableView.backgroundColor = UIColor(red: 0.039, green: 0.039, blue: 0.039, alpha: 1)
        navigationController?.navigationBar.barStyle = .black
        navigationController?.navigationBar.tintColor = UIColor(red: 0.957, green: 0.200, blue: 0.235, alpha: 1)
        guard glucoseGuardTabBarHost == nil else { return }
        navigationController?.setToolbarHidden(true, animated: false)
        let bar = GlucoseGuardTabBar(
            state: glucoseGuardHomeState,
            onToday: { [weak self] in self?.presentGlucoseGuardToday() },
            onLearning: { [weak self] in self?.presentGlucoseGuardLearning() },
            onHealthway: { [weak self] in self?.presentGlucoseGuardHealthway() },
            onAdd: { [weak self] in self?.presentGlucoseGuardLogSheet() }
        )
        let host = UIHostingController(rootView: bar)
        host.view.backgroundColor = .clear
        addChild(host)
        host.view.translatesAutoresizingMaskIntoConstraints = false
        view.addSubview(host.view)
        NSLayoutConstraint.activate([
            host.view.leadingAnchor.constraint(equalTo: view.leadingAnchor),
            host.view.trailingAnchor.constraint(equalTo: view.trailingAnchor),
            host.view.bottomAnchor.constraint(equalTo: view.safeAreaLayoutGuide.bottomAnchor),
            host.view.heightAnchor.constraint(equalToConstant: 86)
        ])
        host.didMove(toParent: self)
        glucoseGuardTabBarHost = host
        tableView.contentInset.bottom = 96
        tableView.verticalScrollIndicatorInsets.bottom = 96
    }

    private func attachGlucoseGuardHomeHUD(to cell: UITableViewCell) {
        cell.backgroundColor = UIColor(red: 0.039, green: 0.039, blue: 0.039, alpha: 1)
        cell.contentView.backgroundColor = UIColor(red: 0.039, green: 0.039, blue: 0.039, alpha: 1)
        if let hudView {
            hudView.isHidden = true
            hudView.alpha = 0
        }
        let hud = GlucoseGuardHomeHUD(
            state: glucoseGuardHomeState,
            onCGM: { [weak self] in self?.handleGlucoseGuardCGMTap() },
            onPump: { [weak self] in self?.presentGlucoseGuardPumpDetail() },
            onLoop: { [weak self] in
                guard let self, let hudView = self.hudView else { return }
                self.showLoopCompletionMessage(hudView.loopCompletionHUD)
            }
        )
        if let host = glucoseGuardHomeHUDHost {
            host.rootView = hud
        } else {
            let host = UIHostingController(rootView: hud)
            host.view.backgroundColor = .clear
            addChild(host)
            host.didMove(toParent: self)
            glucoseGuardHomeHUDHost = host
        }
        guard let hostView = glucoseGuardHomeHUDHost?.view else { return }
        if hostView.superview !== cell.contentView {
            hostView.removeFromSuperview()
            hostView.translatesAutoresizingMaskIntoConstraints = false
            cell.contentView.addSubview(hostView)
            NSLayoutConstraint.activate([
                hostView.leadingAnchor.constraint(equalTo: cell.contentView.leadingAnchor),
                hostView.trailingAnchor.constraint(equalTo: cell.contentView.trailingAnchor),
                hostView.topAnchor.constraint(equalTo: cell.contentView.topAnchor),
                hostView.bottomAnchor.constraint(equalTo: cell.contentView.bottomAnchor)
            ])
        }
        refreshGlucoseGuardHomeHUD()
    }

    private func refreshGlucoseGuardHomeHUD() {
        let state = glucoseGuardHomeState
        state.hasCGM = deviceManager.cgmManager != nil
        state.hasPump = deviceManager.pumpManager != nil
        if let glucose = deviceManager.glucoseStore.latestGlucose {
            state.glucose = deviceManager.displayGlucosePreference.format(glucose.quantity)
            state.trendArrows = deviceManager.glucoseDisplay(for: glucose)?.trendType?.arrows ?? ""
            let formatter = DateFormatter()
            formatter.timeStyle = .short
            state.readingTime = formatter.string(from: glucose.startDate)
        } else {
            state.glucose = "—"
            state.trendArrows = ""
            state.readingTime = ""
        }
        if let iob = currentIOBDescription, !iob.isEmpty {
            state.insulin = iob.contains("IE") || iob.contains("U") ? iob.replacingOccurrences(of: "U", with: "IE") : "\(iob) IE"
        } else {
            state.insulin = "+0.0 IE"
        }
        if let units = deviceManager.doseStore.lastReservoirValue?.unitVolume {
            state.remainingInsulin = units >= 50 ? "50+ IE" : String(format: "%.0f IE", units)
            state.reservoirFraction = min(max(units / 200.0, 0), 1)
        } else {
            state.remainingInsulin = "—"
            state.reservoirFraction = 0
        }
        if let last = deviceManager.loopManager.lastLoopCompleted {
            state.loopConnected = Date().timeIntervalSince(last) < 360
        } else {
            state.loopConnected = false
        }
    }

    private func presentGlucoseGuardDesign<Content: View>(_ root: Content, title: String, light: Bool = false) {
        let hosting = UIHostingController(rootView: root)
        hosting.title = title
        hosting.overrideUserInterfaceStyle = light ? .light : .dark
        hosting.view.backgroundColor = light
            ? UIColor(red: 0.980, green: 0.980, blue: 0.980, alpha: 1)
            : UIColor(red: 0.039, green: 0.039, blue: 0.039, alpha: 1)
        let navigation = UINavigationController(rootViewController: hosting)
        navigation.overrideUserInterfaceStyle = hosting.overrideUserInterfaceStyle
        hosting.navigationItem.rightBarButtonItem = UIBarButtonItem(
            barButtonSystemItem: .done,
            target: navigation,
            action: #selector(UIViewController.dismissWithAnimation)
        )
        present(navigation, animated: true)
    }

    private func handleGlucoseGuardCGMTap() {
        if deviceManager.cgmManager == nil {
            presentGlucoseGuardCGMSourceOverlay()
        } else {
            presentGlucoseGuardCGMDetail()
        }
    }

    private func presentGlucoseGuardCGMSourceOverlay() {
        let sheet = UIAlertController(
            title: NSLocalizedString("Add CGM", comment: "CGM source overlay title"),
            message: NSLocalizedString("LibreLinkUp, Dexcom Share, or another CGM source.", comment: "CGM source overlay subtitle"),
            preferredStyle: .actionSheet
        )
        sheet.addAction(UIAlertAction(title: "LibreLinkUp", style: .default) { [weak self] _ in
            self?.executeHUDTapAction(self?.deviceManager.didTapOnCGMStatus())
        })
        sheet.addAction(UIAlertAction(title: "Dexcom Share", style: .default) { [weak self] _ in
            self?.executeHUDTapAction(self?.deviceManager.didTapOnCGMStatus())
        })
        sheet.addAction(UIAlertAction(
            title: NSLocalizedString("Other CGM", comment: "Open Loop CGM picker"),
            style: .default
        ) { [weak self] _ in
            self?.executeHUDTapAction(self?.deviceManager.didTapOnCGMStatus())
        })
        sheet.addAction(UIAlertAction(title: NSLocalizedString("Cancel", comment: "Cancel CGM source"), style: .cancel))
        present(sheet, animated: true)
    }

    @objc private func presentGlucoseGuardLearning() {
        let unit = deviceManager.displayGlucosePreference.unit
        presentGlucoseGuardDesign(
            GlucoseGuardLearningView(displayUnit: unit) { start, completion in
                self.deviceManager.glucoseStore.getGlucoseSamples(start: start, end: nil) { result in
                    let values: [Double]
                    switch result {
                    case .success(let samples):
                        values = samples.map { $0.quantity.doubleValue(for: .milligramsPerDeciliter) }
                    case .failure:
                        values = []
                    }
                    DispatchQueue.main.async { completion(values) }
                }
            },
            title: NSLocalizedString("Learning", comment: "Learning tab title")
        )
    }

    @objc private func presentGlucoseGuardHealthway() {
        let unit = deviceManager.displayGlucosePreference.unit
        let latest = deviceManager.glucoseStore.latestGlucose
        let current = latest.map { deviceManager.displayGlucosePreference.format($0.quantity) } ?? "—"
        let formatter = DateFormatter()
        formatter.timeStyle = .short
        let readingTime = latest.map { formatter.string(from: $0.startDate) } ?? "—"
        let start = Calendar.current.date(byAdding: .day, value: -7, to: Date()) ?? Date()
        deviceManager.glucoseStore.getGlucoseSamples(start: start, end: nil) { result in
            let values: [Double]
            switch result {
            case .success(let samples):
                values = samples.map { $0.quantity.doubleValue(for: .milligramsPerDeciliter) }
            case .failure:
                values = []
            }
            let average = values.isEmpty ? nil : values.reduce(0, +) / Double(values.count)
            let a1c = average.map { String(format: "%.1f%%", 3.31 + 0.02392 * $0) } ?? "—"
            let avgText: String
            if let average {
                let quantity = HKQuantity(unit: .milligramsPerDeciliter, doubleValue: average)
                avgText = String(format: "%.1f %@", quantity.doubleValue(for: unit), unit.unitString)
            } else {
                avgText = "—"
            }
            let tir: String
            if values.isEmpty {
                tir = "—"
            } else {
                let inRange = values.filter { $0 >= 70 && $0 <= 180 }.count
                tir = String(format: "%.0f%%", (Double(inRange) / Double(values.count)) * 100)
            }
            let spark = Array(values.suffix(48))
            DispatchQueue.main.async {
                self.presentGlucoseGuardDesign(
                    GlucoseGuardHealthwayView(
                        currentGlucose: current,
                        readingTime: readingTime,
                        a1c: a1c,
                        average: avgText,
                        tir: tir,
                        points: spark
                    ),
                    title: NSLocalizedString("Healthway", comment: "Healthway tab title"),
                    light: true
                )
            }
        }
    }

    private func presentGlucoseGuardCGMDetail() {
        let unit = deviceManager.displayGlucosePreference.unit
        let latest = deviceManager.glucoseStore.latestGlucose
        let reading = latest.map { deviceManager.displayGlucosePreference.format($0.quantity) } ?? "— — —"
        let minutes: String
        if let date = latest?.startDate {
            minutes = String(format: "%.0f min", abs(date.timeIntervalSinceNow) / 60)
        } else {
            minutes = "0 min"
        }
        let used = deviceManager.cgmLifecycleProgress?.percentComplete ?? 0
        let remaining = max(0, (1 - used) * 14)
        let periods = [3, 7, 30, 90]
        let group = DispatchGroup()
        var averages: [(Int, String)] = []
        let lock = NSLock()
        for days in periods {
            group.enter()
            let start = Calendar.current.date(byAdding: .day, value: -days, to: Date()) ?? Date()
            deviceManager.glucoseStore.getGlucoseSamples(start: start, end: nil) { result in
                defer { group.leave() }
                let values: [Double]
                switch result {
                case .success(let samples):
                    values = samples.map { $0.quantity.doubleValue(for: .milligramsPerDeciliter) }
                case .failure:
                    values = []
                }
                let text: String
                if values.isEmpty {
                    text = "—"
                } else {
                    let mean = values.reduce(0, +) / Double(values.count)
                    let quantity = HKQuantity(unit: .milligramsPerDeciliter, doubleValue: mean)
                    text = String(format: "%.1f %@", quantity.doubleValue(for: unit), unit.unitString)
                }
                lock.lock()
                averages.append((days, text))
                lock.unlock()
            }
        }
        group.notify(queue: .main) {
            let ordered = averages.sorted { $0.0 < $1.0 }
            self.presentGlucoseGuardDesign(
                GlucoseGuardCGMDetailView(
                    lastReading: reading,
                    minutesAgo: minutes,
                    remainingDays: remaining,
                    sensorLifeDays: 14,
                    averages: ordered,
                    onChooseSource: {
                        self.dismiss(animated: true) {
                            self.presentGlucoseGuardCGMSourceOverlay()
                        }
                    }
                ),
                title: "CGM"
            )
        }
    }

    private func presentGlucoseGuardPumpDetail() {
        let used = deviceManager.pumpLifecycleProgress?.percentComplete ?? 0
        let remaining = max(0, (1 - used) * 3)
        let name = deviceManager.pumpManager?.localizedTitle ?? "Omnipod DASH"
        let basal = hudView?.pumpStatusHUD.basalRateHUD.accessibilityValue ?? "—"
        let remainingInsulin = glucoseGuardHomeState.remainingInsulin
        let currentBolus = glucoseGuardHomeState.insulin
        presentGlucoseGuardDesign(
            GlucoseGuardPumpDetailView(
                pumpName: name,
                remainingDays: remaining,
                remainingHours: remaining * 24,
                podLifeDays: 3,
                basal: basal,
                remainingInsulin: remainingInsulin,
                currentBolus: currentBolus,
                onChangeBolus: {
                    self.dismiss(animated: true) {
                        self.presentBolusScreen()
                    }
                },
                onOpenPump: {
                    self.dismiss(animated: true) {
                        if let pumpView = self.hudView?.pumpStatusHUD {
                            self.executeHUDTapAction(self.deviceManager.didTapOnPumpStatus(pumpView.pumpManagerProvidedHUD))
                        }
                    }
                }
            ),
            title: name
        )
    }
