    // GLUCOSE_GUARD_XCODE_DESIGN host

    private let glucoseGuardHomeState = GlucoseGuardHomeState()
    private var glucoseGuardTabBarHost: UIHostingController<GlucoseGuardTabBar>?

    private func installGlucoseGuardTabBar() {
        overrideUserInterfaceStyle = .dark
        tableView.backgroundColor = UIColor(red: 0.02, green: 0.02, blue: 0.03, alpha: 1)
        navigationController?.navigationBar.barStyle = .black
        navigationController?.navigationBar.tintColor = UIColor(red: 0.957, green: 0.200, blue: 0.235, alpha: 1)
        guard glucoseGuardTabBarHost == nil else { return }
        navigationController?.setToolbarHidden(true, animated: false)
        let bar = GlucoseGuardTabBar(
            state: glucoseGuardHomeState,
            onToday: { [weak self] in self?.presentGlucoseGuardToday() },
            onFood: { [weak self] in self?.presentGlucoseGuardFood() },
            onBolus: { [weak self] in self?.presentBolusScreen() },
            onProfile: { [weak self] in self?.presentGlucoseGuardProfile() },
            onAdd: { [weak self] in self?.presentGlucoseGuardLogSheet() }
        )
        let host = UIHostingController(rootView: bar)
        host.view.backgroundColor = .clear
        addChild(host)
        host.view.translatesAutoresizingMaskIntoConstraints = false
        view.addSubview(host.view)
        NSLayoutConstraint.activate([
            host.view.leadingAnchor.constraint(equalTo: view.leadingAnchor),
            host.view.trailingAnchor.constraint(equalTo: view.trailingAnchor),
            host.view.bottomAnchor.constraint(equalTo: view.safeAreaLayoutGuide.bottomAnchor),
            host.view.heightAnchor.constraint(equalToConstant: 70)
        ])
        host.didMove(toParent: self)
        glucoseGuardTabBarHost = host
        tableView.contentInset.bottom = 118
        tableView.verticalScrollIndicatorInsets.bottom = 118
        GlucoseGuardAlertCenter.shared.install()
        refreshGlucoseGuardReadings()
    }

    private func presentGlucoseGuardDesign<Content: View>(_ root: Content, title: String) {
        let hosting = UIHostingController(rootView: root)
        hosting.title = title
        hosting.overrideUserInterfaceStyle = .dark
        hosting.view.backgroundColor = UIColor(red: 0.02, green: 0.02, blue: 0.03, alpha: 1)
        let navigation = UINavigationController(rootViewController: hosting)
        navigation.overrideUserInterfaceStyle = .dark
        let done = UIBarButtonItem(
            title: NSLocalizedString("Done", comment: "Dismiss Glucose Guard screen"),
            style: .done,
            target: navigation,
            action: #selector(UIViewController.dismissWithAnimation)
        )
        done.tintColor = UIColor(red: 0.957, green: 0.200, blue: 0.235, alpha: 1)
        hosting.navigationItem.leftBarButtonItem = done
        let presenter = presentedViewController ?? self
        presenter.present(navigation, animated: true)
    }

    private func refreshGlucoseGuardReadings() {
        let latest = deviceManager.glucoseStore.latestGlucose
        glucoseGuardHomeState.lastReading = latest.map { deviceManager.displayGlucosePreference.format($0.quantity) } ?? "— — —"
        glucoseGuardHomeState.unitLabel = deviceManager.displayGlucosePreference.unit.unitString
        if let date = latest?.startDate {
            glucoseGuardHomeState.minutesAgo = String(format: "%.0f min", abs(date.timeIntervalSinceNow) / 60)
        } else {
            glucoseGuardHomeState.minutesAgo = "0 min"
        }
        reloadData()
        evaluateGlucoseGuardAlarms()
    }

    private func evaluateGlucoseGuardAlarms() {
        let mgdl = deviceManager.glucoseStore.latestGlucose?.quantity.doubleValue(for: .milligramsPerDeciliter)
        GlucoseGuardAlertCenter.shared.evaluate(mgdl: mgdl, hasCGM: deviceManager.cgmManager != nil)
    }

    private func glucoseGuardDateString(_ date: Date) -> String {
        let formatter = DateFormatter()
        formatter.dateFormat = "d.M.yyyy / HH:mm"
        return formatter.string(from: date)
    }

    private func glucoseGuardLifecycleDates(lifeDays: Double, used: Double) -> (String, String) {
        let life = lifeDays * 24 * 3600
        let inserted = Date().addingTimeInterval(-used * life)
        let expires = inserted.addingTimeInterval(life)
        return (glucoseGuardDateString(inserted), glucoseGuardDateString(expires))
    }

    private func presentGlucoseGuardAccount(service: String) {
        let prefs = GlucoseGuardPreferences.shared
        let isLibre = service == "LibreView"
        presentGlucoseGuardDesign(
            GlucoseGuardAccountView(
                service: service,
                nickname: Binding(
                    get: { isLibre ? prefs.libreNickname : prefs.dexcomNickname },
                    set: { if isLibre { prefs.libreNickname = $0 } else { prefs.dexcomNickname = $0 } }
                ),
                server: Binding(
                    get: { isLibre ? prefs.libreServer : prefs.dexcomServer },
                    set: { if isLibre { prefs.libreServer = $0 } else { prefs.dexcomServer = $0 } }
                ),
                onAdd: { [weak self] in
                    self?.dismiss(animated: true) {
                        self?.executeHUDTapAction(self?.deviceManager.didTapOnCGMStatus())
                    }
                }
            ),
            title: service
        )
    }

    private func presentGlucoseGuardEmergency() {
        presentGlucoseGuardDesign(
            GlucoseGuardEmergencyContactsView(),

    private func handleGlucoseGuardCGMTap() {
        if deviceManager.cgmManager == nil {
            presentGlucoseGuardCGMSourceOverlay()
        } else {
            presentGlucoseGuardCGMDetail()
        }
    }

    private func presentGlucoseGuardCGMSourceOverlay() {
        presentGlucoseGuardConnections()
    }

    @objc private func presentGlucoseGuardLearning() {
        let unit = deviceManager.displayGlucosePreference.unit
        presentGlucoseGuardDesign(
            GlucoseGuardLearningView(displayUnit: unit) { start, completion in
                self.deviceManager.glucoseStore.getGlucoseSamples(start: start, end: nil) { result in
                    let values: [Double]
                    switch result {
                    case .success(let samples):
                        values = samples.map { $0.quantity.doubleValue(for: .milligramsPerDeciliter) }
                    case .failure:
                        values = []
                    }
                    DispatchQueue.main.async { completion(values) }
                }
            },
            title: NSLocalizedString("Learning", comment: "Learning tab title")
        )
    }

    @objc private func presentGlucoseGuardHealthway() {
        let unit = deviceManager.displayGlucosePreference.unit
        let latest = deviceManager.glucoseStore.latestGlucose
        let current = latest.map { deviceManager.displayGlucosePreference.format($0.quantity) } ?? "—"
        let formatter = DateFormatter()
        formatter.timeStyle = .short
        let readingTime = latest.map { formatter.string(from: $0.startDate) } ?? "—"
        let start = Calendar.current.date(byAdding: .day, value: -7, to: Date()) ?? Date()
        deviceManager.glucoseStore.getGlucoseSamples(start: start, end: nil) { result in
            let values: [Double]
            switch result {
            case .success(let samples):
                values = samples.map { $0.quantity.doubleValue(for: .milligramsPerDeciliter) }
            case .failure:
                values = []
            }
            let average = values.isEmpty ? nil : values.reduce(0, +) / Double(values.count)
            let a1c = average.map { String(format: "%.1f%%", 3.31 + 0.02392 * $0) } ?? "—"
            let avgText: String
            if let average {
                let quantity = HKQuantity(unit: .milligramsPerDeciliter, doubleValue: average)
                avgText = String(format: "%.1f %@", quantity.doubleValue(for: unit), unit.unitString)
            } else {
                avgText = "—"
            }
            let tir: String
            if values.isEmpty {
                tir = "—"
            } else {
                let inRange = values.filter { $0 >= 70 && $0 <= 180 }.count
                tir = String(format: "%.0f%%", (Double(inRange) / Double(values.count)) * 100)
            }
            let spark = Array(values.suffix(48))
            DispatchQueue.main.async {
                self.presentGlucoseGuardDesign(
                    GlucoseGuardHealthwayView(
                        currentGlucose: current,
                        readingTime: readingTime,
                        a1c: a1c,
                        average: avgText,
                        tir: tir,
                        points: spark
                    ),
                    title: NSLocalizedString("Healthway", comment: "Healthway tab title")
                )
            }
        }
    }

    private func presentGlucoseGuardCGMDetail() {
        refreshGlucoseGuardReadings()
        let used = deviceManager.cgmLifecycleProgress?.percentComplete ?? 0
        let remaining = max(0, (1 - used) * 14)
        let dates = glucoseGuardLifecycleDates(lifeDays: 14, used: used)
        let title = deviceManager.cgmManager?.localizedTitle ?? NSLocalizedString("CGM", comment: "CGM screen title")
        presentGlucoseGuardDesign(
            GlucoseGuardCGMDetailView(
                state: glucoseGuardHomeState,
                title: title,
                remainingDays: remaining,
                insertionText: dates.0,
                expiryText: dates.1,
                onRefresh: { [weak self] in self?.refreshGlucoseGuardReadings() },
                onNotificationSettings: { [weak self] in self?.presentGlucoseGuardGlucoseSettings() },
                onDeviceSettings: { [weak self] in
                    self?.dismiss(animated: true) { self?.onCGMTapped() }
                },
                onStopSensor: { [weak self] in
                    let testing = self?.deviceManager.cgmManager is TestingCGMManager
                    self?.deviceManager.cgmManager?.delete() { }
                    if testing == true {
                        self?.deviceManager.deleteTestingCGMData()
                    }
                    self?.dismiss(animated: true)
                },
                onDone: { [weak self] in self?.dismiss(animated: true) }
            ),
            title: title
        )
    }

    private var glucoseGuardRemainingInsulin: String {
        guard let units = deviceManager.doseStore.lastReservoirValue?.unitVolume else { return "—" }
        return units >= 50 ? "50+ IE" : String(format: "%.0f IE", units)
    }

    private var glucoseGuardCurrentBolus: String {
        guard let iob = currentIOBDescription, !iob.isEmpty else { return "+0.0 IE" }
        return iob.contains("IE") || iob.contains("U") ? iob.replacingOccurrences(of: "U", with: "IE") : "\(iob) IE"
    }

    private func presentGlucoseGuardPumpDetail() {
        let used = deviceManager.pumpLifecycleProgress?.percentComplete ?? 0
        let remaining = max(0, (1 - used) * 3)
        let name = deviceManager.pumpManager?.localizedTitle ?? "Omnipod DASH"
        let delivery = hudView?.pumpStatusHUD.basalRateHUD.accessibilityValue ?? "—"
        let dates = glucoseGuardLifecycleDates(lifeDays: 3, used: used)
        let details: [(String, String)] = [
            ("Pump", name),
            ("Pump Paired", dates.0),
            ("Pump Expires", dates.1),
            ("Insulin Delivery", delivery),
            ("Insulin Remaining", glucoseGuardRemainingInsulin),
            ("Current Bolus", glucoseGuardCurrentBolus)
        ]
        presentGlucoseGuardDesign(
            GlucoseGuardPumpDetailView(
                pumpName: name,
                remainingDays: remaining,
                remainingHours: remaining * 24,
                insulinDelivery: delivery,
                remainingInsulin: glucoseGuardRemainingInsulin,
                pairedText: dates.0,
                expiryText: dates.1,
                details: details,
                onSuspend: { [weak self] in
                    self?.dismiss(animated: true) { self?.onPumpTapped() }
                },
                onReplace: { [weak self] in
                    self?.dismiss(animated: true) { self?.onPumpTapped() }
                },
                onDone: { [weak self] in self?.dismiss(animated: true) }
            ),
            title: name
        )
    }
