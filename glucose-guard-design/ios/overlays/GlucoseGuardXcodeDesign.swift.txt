private enum GlucoseGuardTheme {
    static let red = Color(red: 0.957, green: 0.200, blue: 0.235)
    static let redDeep = Color(red: 0.72, green: 0.08, blue: 0.14)
    static let green = Color(red: 0.22, green: 0.87, blue: 0.52)
    static let ink = Color(red: 0.02, green: 0.02, blue: 0.03)
    static let mist = Color(red: 0.07, green: 0.07, blue: 0.09)
    static let card = Color(red: 0.11, green: 0.11, blue: 0.13)
    static let muted = Color(red: 0.62, green: 0.62, blue: 0.66)
    static let hairline = Color.white.opacity(0.10)
    static let insulin = Color(red: 1.0, green: 0.67, blue: 0.22)
    static let cardRadius: CGFloat = 26
}

final class GlucoseGuardHomeState: ObservableObject {
    @Published var selectedTab = 0
}

struct GlucoseGuardMark: View {
    var size: CGFloat = 36

    var body: some View {
        Image("glucose_guard_mark", bundle: Bundle(for: StatusBarHUDView.self))
            .resizable()
            .scaledToFit()
            .frame(width: size, height: size)
            .clipShape(RoundedRectangle(cornerRadius: size * 0.22, style: .continuous))
            .shadow(color: GlucoseGuardTheme.red.opacity(0.35), radius: 8, y: 2)
    }
}

struct GlucoseGuardBackdrop: View {
    var body: some View {
        ZStack {
            GlucoseGuardTheme.ink
            RadialGradient(
                colors: [GlucoseGuardTheme.red.opacity(0.22), Color.clear],
                center: .topLeading,
                startRadius: 20,
                endRadius: 420
            )
            RadialGradient(
                colors: [GlucoseGuardTheme.green.opacity(0.08), Color.clear],
                center: .bottomTrailing,
                startRadius: 10,
                endRadius: 340
            )
        }
        .ignoresSafeArea()
    }
}

struct GlucoseGuardGlass<Content: View>: View {
    var radius: CGFloat = GlucoseGuardTheme.cardRadius
    let content: Content

    init(radius: CGFloat = GlucoseGuardTheme.cardRadius, @ViewBuilder content: () -> Content) {
        self.radius = radius
        self.content = content()
    }

    var body: some View {
        content
            .padding(18)
            .background(
                RoundedRectangle(cornerRadius: radius, style: .continuous)
                    .fill(GlucoseGuardTheme.card.opacity(0.92))
                    .overlay(
                        RoundedRectangle(cornerRadius: radius, style: .continuous)
                            .stroke(LinearGradient(
                                colors: [Color.white.opacity(0.18), Color.white.opacity(0.04)],
                                startPoint: .topLeading,
                                endPoint: .bottomTrailing
                            ), lineWidth: 1)
                    )
            )
            .shadow(color: Color.black.opacity(0.45), radius: 18, y: 10)
    }
}

struct GlucoseGuardProgressBar: View {
    let progress: Double
    var tint: Color = GlucoseGuardTheme.red

    var body: some View {
        GeometryReader { geo in
            ZStack(alignment: .leading) {
                Capsule().fill(Color.white.opacity(0.08))
                Capsule()
                    .fill(LinearGradient(colors: [tint, tint.opacity(0.7)], startPoint: .leading, endPoint: .trailing))
                    .frame(width: max(10, geo.size.width * CGFloat(min(max(progress, 0), 1))))
                    .shadow(color: tint.opacity(0.55), radius: 6, y: 0)
            }
        }
        .frame(height: 7)
    }
}

struct GlucoseGuardPrimaryButton: View {

                )

}

struct GlucoseGuardTabBar: View {
    @ObservedObject var state: GlucoseGuardHomeState
    var onToday: () -> Void
    var onFood: () -> Void
    var onBolus: () -> Void
    var onProfile: () -> Void
    var onAdd: () -> Void

    var body: some View {
        ZStack(alignment: .topTrailing) {
            VStack(spacing: 0) {
                Rectangle()
                    .fill(Color.white.opacity(0.10))
                    .frame(height: 0.5)
                HStack(alignment: .bottom, spacing: 0) {
                    tab("Today", index: 0, systemName: "house", filled: "house.fill", action: onToday)
                    tab("Food", index: 1, systemName: "fork.knife", filled: "fork.knife", action: onFood)
                    tab("Bolus", index: 2, systemName: "bolt", filled: "bolt.fill", action: onBolus)
                    tab("Profile", index: 3, systemName: "person", filled: "person.fill", action: onProfile)
                }
                .padding(.horizontal, 2)
                .padding(.top, 8)
                .padding(.bottom, 4)
            }
            .background(GlucoseGuardTheme.ink.opacity(0.96))

            Button(action: onAdd) {
                Image(systemName: "plus")
                    .font(.system(size: 20, weight: .medium))
                    .foregroundColor(.white)
                    .frame(width: 48, height: 48)
                    .background(
                        RoundedRectangle(cornerRadius: 14, style: .continuous)
                            .fill(Color(red: 0.18, green: 0.18, blue: 0.20))
                            .overlay(
                                RoundedRectangle(cornerRadius: 14, style: .continuous)
                                    .stroke(Color.white.opacity(0.14), lineWidth: 1)
                            )
                            .shadow(color: Color.black.opacity(0.45), radius: 10, y: 4)
                    )
            }
            .buttonStyle(.plain)
            .padding(.trailing, 16)
            .offset(y: -34)
            .accessibilityLabel(Text("Log bolus, glucose, or food"))
        }
        .frame(height: 70)
    }

    private func tab(_ title: String, index: Int, systemName: String, filled: String, action: @escaping () -> Void) -> some View {
        let selected = state.selectedTab == index
        return Button {
            state.selectedTab = index
            action()
        } label: {
            VStack(spacing: 5) {
                ZStack {
                    if selected {
                        Circle()
                            .fill(GlucoseGuardTheme.red.opacity(0.38))
                            .frame(width: 34, height: 34)
                            .blur(radius: 9)
                    }
                    Image(systemName: selected ? filled : systemName)
                        .font(.system(size: 20, weight: selected ? .semibold : .regular))
                }
                .frame(height: 24)
                Text(title)
                    .font(.system(size: 10, weight: .semibold))
                Capsule()
                    .fill(selected ? GlucoseGuardTheme.red : Color.clear)
                    .frame(width: 22, height: 2)
            }
            .foregroundColor(selected ? GlucoseGuardTheme.red : Color.white)
            .frame(maxWidth: .infinity)
        }
        .buttonStyle(.plain)
    }
}

                        Text(NSLocalizedString("Current bolus", comment: "Pump bolus card").uppercased())
                            .font(.system(size: 10, weight: .bold))
                            .tracking(1.1)

                        .foregroundColor(GlucoseGuardTheme.muted)
                        .frame(maxWidth: .infinity)
                        .padding(.vertical, 12)
                }
            }
        }
    }

    private var podExpiryText: String {
        let days = Int(remainingDays)
        let hours = Int((remainingDays - Double(days)) * 24)
        if days <= 0 {
            return String(format: NSLocalizedString("%d hours", comment: "Pod hours remaining"), max(hours, Int(remainingHours)))
        }
        return String(format: NSLocalizedString("%d days %d hours", comment: "Pod days and hours remaining"), days, hours)
    }

struct GlucoseGuardSparkline: View {
    let points: [Double]

    var body: some View {
        GeometryReader { geo in
            let values = points.isEmpty ? [90, 110, 95, 130, 100, 118, 102] : points
            let minV = min(values.min() ?? 0, 70) - 10
            let maxV = max(values.max() ?? 1, 180) + 10
            let span = max(maxV - minV, 1)
            Path { path in
                path.move(to: CGPoint(x: 0, y: geo.size.height * (1 - CGFloat((70 - minV) / span))))
                path.addLine(to: CGPoint(x: geo.size.width, y: geo.size.height * (1 - CGFloat((70 - minV) / span))))
            }
            .stroke(Color.white.opacity(0.12), style: StrokeStyle(lineWidth: 1, dash: [4, 4]))
            Path { path in
                path.move(to: CGPoint(x: 0, y: geo.size.height * (1 - CGFloat((180 - minV) / span))))
                path.addLine(to: CGPoint(x: geo.size.width, y: geo.size.height * (1 - CGFloat((180 - minV) / span))))
            }
            .stroke(GlucoseGuardTheme.green.opacity(0.28), style: StrokeStyle(lineWidth: 1, dash: [4, 4]))
            Path { path in
                for (index, value) in values.enumerated() {
                    let x = geo.size.width * CGFloat(index) / CGFloat(max(values.count - 1, 1))
                    let y = geo.size.height * (1 - CGFloat((value - minV) / span))
                    if index == 0 {
                        path.move(to: CGPoint(x: x, y: geo.size.height))
                        path.addLine(to: CGPoint(x: x, y: y))
                    } else {
                        path.addLine(to: CGPoint(x: x, y: y))
                    }
                }
                path.addLine(to: CGPoint(x: geo.size.width, y: geo.size.height))
                path.closeSubpath()
            }
            .fill(LinearGradient(colors: [GlucoseGuardTheme.red.opacity(0.38), Color.clear], startPoint: .top, endPoint: .bottom))
            Path { path in
                for (index, value) in values.enumerated() {
                    let x = geo.size.width * CGFloat(index) / CGFloat(max(values.count - 1, 1))
                    let y = geo.size.height * (1 - CGFloat((value - minV) / span))
                    if index == 0 {
                        path.move(to: CGPoint(x: x, y: y))
                    } else {
                        path.addLine(to: CGPoint(x: x, y: y))
                    }
                }
            }
            .stroke(GlucoseGuardTheme.red, lineWidth: 2.4)
            .shadow(color: GlucoseGuardTheme.red.opacity(0.55), radius: 8)
        }
    }
}
