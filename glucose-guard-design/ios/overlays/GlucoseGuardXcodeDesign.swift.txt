private enum GlucoseGuardTheme {
    static let red = Color(red: 0.957, green: 0.200, blue: 0.235)
    static let green = Color(red: 0.133, green: 0.773, blue: 0.369)
    static let ink = Color(red: 0.039, green: 0.039, blue: 0.039)
    static let surface = Color(red: 0.090, green: 0.090, blue: 0.090)
    static let card = Color(red: 0.173, green: 0.173, blue: 0.173)
    static let muted = Color(red: 0.639, green: 0.639, blue: 0.639)
    static let paper = Color(red: 0.980, green: 0.980, blue: 0.980)
    static let barBlue = Color(red: 0.23, green: 0.51, blue: 0.96)
    static let insulin = Color(red: 1.0, green: 0.62, blue: 0.11)
    static let cardRadius: CGFloat = 18
}

final class GlucoseGuardHomeState: ObservableObject {
    @Published var hasCGM = false
    @Published var hasPump = false
    @Published var glucose = "—"
    @Published var trendArrows = ""
    @Published var readingTime = ""
    @Published var insulin = "+0.0 IE"
    @Published var reservoirFraction: Double = 0
    @Published var remainingInsulin = "—"
    @Published var loopConnected = false
    @Published var selectedTab = 0
}

struct GlucoseGuardMark: View {
    var size: CGFloat = 36

    var body: some View {
        Image("glucose_guard_mark", bundle: Bundle(for: StatusBarHUDView.self))
            .resizable()
            .scaledToFit()
            .frame(width: size, height: size)
            .clipShape(RoundedRectangle(cornerRadius: size * 0.22, style: .continuous))
    }
}

struct GlucoseGuardProgressBar: View {
    let progress: Double
    var tint: Color = GlucoseGuardTheme.barBlue

    var body: some View {
        GeometryReader { geo in
            ZStack(alignment: .leading) {
                Capsule().fill(Color.white.opacity(0.12))
                Capsule()
                    .fill(tint)
                    .frame(width: max(6, geo.size.width * CGFloat(min(max(progress, 0), 1))))
            }
        }
        .frame(height: 6)
    }
}

struct GlucoseGuardHomeHUD: View {
    @ObservedObject var state: GlucoseGuardHomeState
    var onCGM: () -> Void
    var onPump: () -> Void
    var onLoop: () -> Void

    var body: some View {
        HStack(spacing: 10) {
            Button(action: onCGM) {
                Group {
                    if state.hasCGM {
                        VStack(alignment: .leading, spacing: 2) {
                            HStack(alignment: .firstTextBaseline, spacing: 4) {
                                Text(state.glucose)
                                    .font(.system(size: 18, weight: .bold, design: .rounded))
                                Text(state.trendArrows)
                                    .font(.system(size: 20, weight: .heavy))
                                    .foregroundColor(GlucoseGuardTheme.red)
                            }
                            Text(state.readingTime)
                                .font(.caption2.weight(.semibold))
                                .foregroundColor(GlucoseGuardTheme.muted)
                        }
                    } else {
                        HStack(spacing: 6) {
                            Text("Add CGM")
                                .font(.subheadline.weight(.semibold))
                            Image(systemName: "plus")
                                .font(.system(size: 13, weight: .bold))
                        }
                    }
                }
                .foregroundColor(.white)
                .padding(.horizontal, 14)
                .padding(.vertical, 10)
                .frame(maxWidth: .infinity, minHeight: 58)
                .background(GlucoseGuardTheme.card)
                .clipShape(Capsule())
            }
            .buttonStyle(.plain)
            .accessibilityLabel(Text(state.hasCGM ? "\(state.glucose) \(state.trendArrows)" : "Add CGM"))

            Button(action: onLoop) {
                ZStack {
                    Circle()
                        .stroke(state.loopConnected ? GlucoseGuardTheme.green : Color.white.opacity(0.92), lineWidth: 5)
                        .frame(width: 58, height: 58)
                    GlucoseGuardMark(size: 42)
                }
            }
            .buttonStyle(.plain)
            .accessibilityLabel(Text(state.loopConnected ? "Closed loop connected" : "Loop waiting"))

            Button(action: onPump) {
                HStack(spacing: 8) {
                    VStack(alignment: .leading, spacing: 5) {
                        Text(state.insulin)
                            .font(.system(size: 16, weight: .bold, design: .rounded))
                        GlucoseGuardProgressBar(progress: state.reservoirFraction, tint: GlucoseGuardTheme.red)
                            .frame(width: 54)
                    }
                    Image(systemName: "syringe.fill")
                        .font(.system(size: 16, weight: .semibold))
                        .foregroundColor(GlucoseGuardTheme.red)
                }
                .foregroundColor(.white)
                .padding(.horizontal, 14)
                .padding(.vertical, 10)
                .frame(maxWidth: .infinity, minHeight: 58)
                .background(GlucoseGuardTheme.card)
                .clipShape(RoundedRectangle(cornerRadius: 18, style: .continuous))
            }
            .buttonStyle(.plain)
            .accessibilityLabel(Text(state.hasPump ? state.insulin : "Insulin pump"))
        }
        .padding(.horizontal, 12)
        .padding(.vertical, 8)
        .background(GlucoseGuardTheme.ink)
    }
}

struct GlucoseGuardTabBar: View {
    @ObservedObject var state: GlucoseGuardHomeState
    var onToday: () -> Void
    var onLearning: () -> Void
    var onHealthway: () -> Void
    var onAdd: () -> Void

    var body: some View {
        ZStack(alignment: .top) {
            GlucoseGuardTheme.ink.ignoresSafeArea(edges: .bottom)
            HStack(alignment: .bottom, spacing: 0) {
                tab("Today", selected: state.selectedTab == 0, action: {
                    state.selectedTab = 0
                    onToday()
                }) {
                    GlucoseGuardTodayIcon()
                }
                tab("Learning", selected: state.selectedTab == 1, action: {
                    state.selectedTab = 1
                    onLearning()
                }) {
                    GlucoseGuardLearningIcon()
                }
                tab("Healthway", selected: state.selectedTab == 2, action: {
                    state.selectedTab = 2
                    onHealthway()
                }) {
                    GlucoseGuardHealthwayIcon()
                }
                Button(action: onAdd) {
                    VStack(spacing: 4) {
                        ZStack {
                            Circle()
                                .fill(GlucoseGuardTheme.red)
                                .frame(width: 62, height: 62)
                                .shadow(color: GlucoseGuardTheme.red.opacity(0.4), radius: 10, y: 4)
                            Image(systemName: "plus")
                                .font(.system(size: 28, weight: .bold))
                                .foregroundColor(.white)
                        }
                        .offset(y: -10)
                        Text("Log")
                            .font(.caption2.weight(.semibold))
                            .foregroundColor(.white)
                    }
                    .frame(maxWidth: .infinity)
                }
                .buttonStyle(.plain)
                .accessibilityLabel(Text("Log bolus, glucose, or food"))
            }
            .padding(.horizontal, 8)
            .padding(.top, 8)
            .padding(.bottom, 10)
        }
        .frame(height: 86)
    }

    private func tab<Icon: View>(_ title: String, selected: Bool, action: @escaping () -> Void, @ViewBuilder icon: () -> Icon) -> some View {
        Button(action: action) {
            VStack(spacing: 5) {
                icon()
                    .frame(height: 22)
                Text(title)
                    .font(.caption2.weight(.semibold))
            }
            .foregroundColor(selected ? GlucoseGuardTheme.red : .white)
            .frame(maxWidth: .infinity)
        }
        .buttonStyle(.plain)
    }
}

struct GlucoseGuardScreenChrome<Content: View>: View {
    let title: String
    let content: Content

    init(_ title: String, @ViewBuilder content: () -> Content) {
        self.title = title
        self.content = content()
    }

    var body: some View {
        ZStack {
            GlucoseGuardTheme.ink.ignoresSafeArea()
            ScrollView {
                VStack(alignment: .leading, spacing: 16) {
                    HStack(spacing: 10) {
                        GlucoseGuardMark(size: 36)
                        Text(title)
                            .font(.title2.bold())
                            .foregroundColor(.white)
                        Spacer()
                    }
                    content
                }
                .padding(20)
            }
        }
    }
}

    let onChangeBolus: () -> Void
    let onOpenPump: () -> Void

            VStack(alignment: .leading, spacing: 14) {
                VStack(spacing: 16) {
                    HStack {
                        Text(pumpName)

                        Spacer()
                        Image(systemName: "antenna.radiowaves.left.and.right")
                            .foregroundColor(GlucoseGuardTheme.barBlue)
                    }
                    ZStack {
                        Circle()
                            .fill(Color.white.opacity(0.08))
                            .frame(width: 96, height: 96)
                        Image(systemName: "oval.portrait.fill")
                            .font(.system(size: 54))
                            .foregroundColor(.white)
                    }
                    VStack(alignment: .leading, spacing: 6) {
                        Text(NSLocalizedString("Pod expires in", comment: "Pump remaining life label"))
                            .font(.caption)
                            .foregroundColor(GlucoseGuardTheme.muted)
                        Text(podExpiryText)
                            .font(.title3.bold())
                            .foregroundColor(.white)
                        GlucoseGuardProgressBar(progress: 1 - (remainingDays / max(podLifeDays, 1)))
                    }
                    HStack {
                        VStack(alignment: .leading, spacing: 4) {
                            Text(NSLocalizedString("Planned basal", comment: "Pump basal label"))
                                .font(.caption)
                                .foregroundColor(GlucoseGuardTheme.muted)
                            Text(basal)
                                .font(.headline)
                                .foregroundColor(.white)
                        }
                        Spacer()
                        VStack(alignment: .trailing, spacing: 4) {
                            Text(NSLocalizedString("Remaining insulin", comment: "Pump reservoir label"))
                                .font(.caption)
                                .foregroundColor(GlucoseGuardTheme.muted)
                            HStack(spacing: 6) {
                                Image(systemName: "drop.fill")
                                    .foregroundColor(GlucoseGuardTheme.insulin)
                                Text(remainingInsulin)
                                    .font(.title3.bold())
                                    .foregroundColor(.white)
                            }
                        }
                    }
                }
                .padding(16)
                .background(GlucoseGuardTheme.card)
                .clipShape(RoundedRectangle(cornerRadius: GlucoseGuardTheme.cardRadius, style: .continuous))

                VStack(alignment: .leading, spacing: 10) {
                    Text(NSLocalizedString("Current bolus", comment: "Pump bolus card"))
                        .font(.headline)
                        .foregroundColor(.white)
                    Text(currentBolus)
                        .font(.title2.bold())
                        .foregroundColor(.white)
                    Button(action: onChangeBolus) {
                        Text(NSLocalizedString("Change bolus", comment: "Change current bolus"))
                            .font(.headline)
                            .foregroundColor(.white)
                            .frame(maxWidth: .infinity)
                            .padding(.vertical, 14)
                            .background(GlucoseGuardTheme.red)
                            .clipShape(RoundedRectangle(cornerRadius: 14, style: .continuous))
                    }
                }
                .padding(16)
                .background(GlucoseGuardTheme.card)
                .clipShape(RoundedRectangle(cornerRadius: GlucoseGuardTheme.cardRadius, style: .continuous))

                Button(action: onOpenPump) {
                    Text(NSLocalizedString("Open pump settings", comment: "Open Loop pump UI"))
                        .font(.subheadline.weight(.semibold))
                        .foregroundColor(.white)
                        .frame(maxWidth: .infinity)
                        .padding(.vertical, 12)
                }
            }
        }
    }

    private var podExpiryText: String {
        let days = Int(remainingDays)
        let hours = Int((remainingDays - Double(days)) * 24)
        if days <= 0 {
            return String(format: NSLocalizedString("%d hours", comment: "Pod hours remaining"), max(hours, Int(remainingHours)))
        }
        return String(format: NSLocalizedString("%d days %d hours", comment: "Pod days and hours remaining"), days, hours)
    }
}

                        .foregroundColor(GlucoseGuardTheme.ink)
                    Spacer()
                }

                    Text("Day").tag(1)
                    Text("Week").tag(7)

                VStack(alignment: .leading, spacing: 12) {
                    HStack {
                        Text("Glucose / \(readingTime)")
                            .foregroundColor(GlucoseGuardTheme.muted)

                .font(.caption)

            Text(value)
                .font(.headline)
                .foregroundColor(GlucoseGuardTheme.red)
        }
        .frame(maxWidth: .infinity, alignment: .leading)
    }
}

struct GlucoseGuardSparkline: View {
    let points: [Double]

    var body: some View {
        GeometryReader { geo in
            let values = points.isEmpty ? [90, 110, 95, 130, 100, 118, 102] : points
            let minV = min(values.min() ?? 0, 70) - 10
            let maxV = max(values.max() ?? 1, 180) + 10
            let span = max(maxV - minV, 1)
            Path { path in
                path.move(to: CGPoint(x: 0, y: geo.size.height * (1 - CGFloat((70 - minV) / span))))
                path.addLine(to: CGPoint(x: geo.size.width, y: geo.size.height * (1 - CGFloat((70 - minV) / span))))
            }
            .stroke(Color.blue.opacity(0.35), style: StrokeStyle(lineWidth: 1, dash: [4, 4]))
            Path { path in
                path.move(to: CGPoint(x: 0, y: geo.size.height * (1 - CGFloat((180 - minV) / span))))
                path.addLine(to: CGPoint(x: geo.size.width, y: geo.size.height * (1 - CGFloat((180 - minV) / span))))
            }
            .stroke(Color.blue.opacity(0.35), style: StrokeStyle(lineWidth: 1, dash: [4, 4]))
            Path { path in
                for (index, value) in values.enumerated() {
                    let x = geo.size.width * CGFloat(index) / CGFloat(max(values.count - 1, 1))
                    let y = geo.size.height * (1 - CGFloat((value - minV) / span))
                    if index == 0 {
                        path.move(to: CGPoint(x: x, y: geo.size.height))
                        path.addLine(to: CGPoint(x: x, y: y))
                    } else {
                        path.addLine(to: CGPoint(x: x, y: y))
                    }
                }
                path.addLine(to: CGPoint(x: geo.size.width, y: geo.size.height))
                path.closeSubpath()
            }
            .fill(GlucoseGuardTheme.red.opacity(0.22))
            Path { path in
                for (index, value) in values.enumerated() {
                    let x = geo.size.width * CGFloat(index) / CGFloat(max(values.count - 1, 1))
                    let y = geo.size.height * (1 - CGFloat((value - minV) / span))
                    if index == 0 {
                        path.move(to: CGPoint(x: x, y: y))
                    } else {
                        path.addLine(to: CGPoint(x: x, y: y))
                    }
                }
            }
            .stroke(GlucoseGuardTheme.red, lineWidth: 2)
        }
    }
}
