private enum GlucoseGuardTheme {
    static let red = Color(red: 0.957, green: 0.200, blue: 0.235)
    static let redDeep = Color(red: 0.72, green: 0.08, blue: 0.14)
    static let green = Color(red: 0.22, green: 0.87, blue: 0.52)
    static let ink = Color(red: 0.02, green: 0.02, blue: 0.03)
    static let mist = Color(red: 0.07, green: 0.07, blue: 0.09)
    static let card = Color(red: 0.11, green: 0.11, blue: 0.13)
    static let muted = Color(red: 0.62, green: 0.62, blue: 0.66)
    static let hairline = Color.white.opacity(0.10)
    static let insulin = Color(red: 1.0, green: 0.67, blue: 0.22)
    static let cardRadius: CGFloat = 26
}

final class GlucoseGuardHomeState: ObservableObject {
    @Published var selectedTab = 0
    @Published var lastReading = "— — —"
    @Published var minutesAgo = "0 min"
    @Published var unitLabel = "mmol/L"
}

struct GlucoseGuardMark: View {
    var size: CGFloat = 36

    var body: some View {
        Image("glucose_guard_mark", bundle: Bundle(for: StatusBarHUDView.self))
            .resizable()
            .scaledToFit()
            .frame(width: size, height: size)
            .clipShape(RoundedRectangle(cornerRadius: size * 0.22, style: .continuous))
            .shadow(color: GlucoseGuardTheme.red.opacity(0.35), radius: 8, y: 2)
    }
}

struct GlucoseGuardBackdrop: View {
    var body: some View {
        ZStack {
            GlucoseGuardTheme.ink
            RadialGradient(
                colors: [GlucoseGuardTheme.red.opacity(0.22), Color.clear],
                center: .topLeading,
                startRadius: 20,
                endRadius: 420
            )
            RadialGradient(
                colors: [GlucoseGuardTheme.green.opacity(0.08), Color.clear],
                center: .bottomTrailing,
                startRadius: 10,
                endRadius: 340
            )
        }
        .ignoresSafeArea()
    }
}

struct GlucoseGuardGlass<Content: View>: View {
    var radius: CGFloat = GlucoseGuardTheme.cardRadius
    let content: Content

    init(radius: CGFloat = GlucoseGuardTheme.cardRadius, @ViewBuilder content: () -> Content) {
        self.radius = radius
        self.content = content()
    }

    var body: some View {
        content
            .padding(18)
            .background(
                RoundedRectangle(cornerRadius: radius, style: .continuous)
                    .fill(GlucoseGuardTheme.card.opacity(0.92))
                    .overlay(
                        RoundedRectangle(cornerRadius: radius, style: .continuous)
                            .stroke(LinearGradient(
                                colors: [Color.white.opacity(0.18), Color.white.opacity(0.04)],
                                startPoint: .topLeading,
                                endPoint: .bottomTrailing
                            ), lineWidth: 1)
                    )
            )
            .shadow(color: Color.black.opacity(0.45), radius: 18, y: 10)
    }
}

struct GlucoseGuardProgressBar: View {

    var body: some View {

                Capsule()
                    .fill(LinearGradient(colors: [tint, tint.opacity(0.7)], startPoint: .leading, endPoint: .trailing))
                    .frame(width: max(10, geo.size.width * CGFloat(min(max(progress, 0), 1))))
                    .shadow(color: tint.opacity(0.55), radius: 6, y: 0)
            }
        }
        .frame(height: 7)
    }
}

struct GlucoseGuardTabBar: View {
    @ObservedObject var state: GlucoseGuardHomeState
    var bottomInset: CGFloat = 34
    var onToday: () -> Void
    var onFood: () -> Void
    var onBolus: () -> Void
    var onProfile: () -> Void
    var onAdd: () -> Void

    var body: some View {
        VStack(spacing: 0) {
            HStack {
                Spacer()
                Button(action: onAdd) {
                    Image(systemName: "plus")
                        .font(.system(size: 22, weight: .medium))
                        .foregroundColor(.white)
                        .frame(width: 48, height: 48)
                        .background(
                            RoundedRectangle(cornerRadius: 14, style: .continuous)
                                .fill(Color(red: 0.18, green: 0.18, blue: 0.20))
                                .overlay(
                                    RoundedRectangle(cornerRadius: 14, style: .continuous)
                                        .stroke(Color.white.opacity(0.14), lineWidth: 1)
                                )
                        )
                }
                .buttonStyle(.plain)
                .accessibilityLabel(Text("Log bolus, glucose, or food"))
            }
            .padding(.horizontal, 16)
            .padding(.top, 8)
            .padding(.bottom, 4)

            HStack(spacing: 0) {
                tab("Today", index: 0, systemName: "house", filled: "house.fill", persists: true, action: onToday)
                tab("Food", index: 1, systemName: "fork.knife", filled: "fork.knife", persists: false, action: onFood)
                tab("Bolus", index: 2, systemName: "bolt", filled: "bolt.fill", persists: false, action: onBolus)
                tab("Profile", index: 3, systemName: "person", filled: "person.fill", persists: true, action: onProfile)
            }
            .padding(.horizontal, 6)
            .padding(.bottom, 6)

            Color.clear.frame(height: max(bottomInset, 8))
        }
        .frame(maxWidth: .infinity)
        .background(GlucoseGuardTheme.ink.ignoresSafeArea(edges: .bottom))
        .ignoresSafeArea(edges: .bottom)
        .overlay(alignment: .top) {
            Rectangle()
                .fill(Color.white.opacity(0.10))
                .frame(height: 0.5)
        }
    }

    private func tab(_ title: String, index: Int, systemName: String, filled: String, persists: Bool, action: @escaping () -> Void) -> some View {
        let selected = state.selectedTab == index
        return Button {
            if persists {
                state.selectedTab = index
            }
            action()
        } label: {
            VStack(spacing: 4) {
                ZStack {
                    if selected {
                        Circle()
                            .fill(GlucoseGuardTheme.red.opacity(0.35))
                            .frame(width: 28, height: 28)
                            .blur(radius: 7)
                    }
                    Image(systemName: selected ? filled : systemName)
                        .font(.system(size: 18, weight: selected ? .semibold : .regular))
                }
                .frame(height: 22)
                Text(title)
                    .font(.system(size: 11, weight: .semibold))
                    .lineLimit(1)
                    .minimumScaleFactor(0.75)
                Capsule()
                    .fill(selected ? GlucoseGuardTheme.red : Color.clear)
                    .frame(width: 18, height: 2)
            }
            .foregroundColor(selected ? GlucoseGuardTheme.red : Color.white)
            .frame(maxWidth: .infinity)
            .padding(.vertical, 4)
        }
        .buttonStyle(.plain)
    }
}

    let remainingInsulin: String
    let pairedText: String
    let expiryText: String
    let details: [(String, String)]
    let onSuspend: () -> Void
    let onReplace: () -> Void
    let onDone: () -> Void

                                .tracking(1.0)
                                .foregroundColor(GlucoseGuardTheme.muted)
                            Text(podExpiryText)
                                .font(.system(size: 26, weight: .bold, design: .rounded))
                                .foregroundColor(.white)
                        }
                    }
                }

                GlucoseGuardGlass {
                    HStack {
                        VStack(alignment: .leading, spacing: 4) {
                            Text(NSLocalizedString("Insulin Delivery", comment: "Pump basal card").uppercased())
                                .font(.system(size: 10, weight: .bold))
                                .tracking(0.8)
                                .foregroundColor(GlucoseGuardTheme.muted)
                            Text(insulinDelivery)
                                .font(.title3.weight(.semibold))
                                .foregroundColor(.white)
                        }
                        Spacer()
                        VStack(alignment: .trailing, spacing: 4) {
                            Text(NSLocalizedString("Insulin Remaining", comment: "Pump reservoir card").uppercased())
                                .font(.system(size: 10, weight: .bold))
                                .tracking(0.8)
                                .foregroundColor(GlucoseGuardTheme.muted)
                            Text(remainingInsulin)
                                .font(.title3.bold())
                                .foregroundColor(.white)
                        }
                    }
                }

                GlucoseGuardGlass {
                    VStack(alignment: .leading, spacing: 0) {
                        Text(NSLocalizedString("Activity", comment: "Pump activity section"))
                            .font(.headline)
                            .foregroundColor(.white)
                            .padding(.bottom, 8)
                        GlucoseGuardNavRow(
                            title: NSLocalizedString("Suspend Insulin Delivery", comment: "Suspend pump"),
                            destructive: true,
                            action: onSuspend
                        )
                    }
                }

                GlucoseGuardGlass {
                    VStack(spacing: 0) {
                        detail("Pump Paired", pairedText)
                        Divider().overlay(GlucoseGuardTheme.hairline).padding(.vertical, 8)
                        detail("Pump Expires", expiryText)
                        Divider().overlay(GlucoseGuardTheme.hairline).padding(.vertical, 8)
                        GlucoseGuardNavRow(
                            title: NSLocalizedString("Device Details", comment: "Pump device details"),
                            action: { showDetails = true }
                        )
                    }
                }

                Button(action: onReplace) {
                    Text(NSLocalizedString("Replace Pump", comment: "Replace the active pump"))
                        .font(.headline)
                        .foregroundColor(GlucoseGuardTheme.red)
                        .frame(maxWidth: .infinity)
                        .padding(.vertical, 16)
                }
                .buttonStyle(.plain)

                GlucoseGuardPrimaryButton(
                    title: NSLocalizedString("Done", comment: "Dismiss pump screen"),
                    action: onDone
                )
            }
        }
        .sheet(isPresented: $showDetails) {
            ZStack {
                GlucoseGuardBackdrop()
                VStack(alignment: .leading, spacing: 14) {
                    Text(NSLocalizedString("Device Details", comment: "Pump details sheet"))
                        .font(.title2.bold())
                        .foregroundColor(.white)
                    ForEach(details, id: \.0) { item in
                        HStack {
                            Text(item.0).foregroundColor(GlucoseGuardTheme.muted)
                            Spacer()
                            Text(item.1).foregroundColor(.white)
                        }
                        .font(.subheadline)
                    }
                    Spacer()
                    GlucoseGuardPrimaryButton(title: NSLocalizedString("Done", comment: "Close device details"), action: { showDetails = false })
                }
                .padding(24)
            }
        }
    }

    private func detail(_ title: String, _ value: String) -> some View {
        HStack {
            Text(title).foregroundColor(GlucoseGuardTheme.muted)
            Spacer()
            Text(value).foregroundColor(.white)
        }
        .font(.subheadline)
    }

    private var podExpiryText: String {
        let days = Int(remainingDays)
        let hours = Int((remainingDays - Double(days)) * 24)
        if days <= 0 {
            return String(format: NSLocalizedString("%d Hours", comment: "Pump hours remaining"), max(hours, Int(remainingHours)))
        }
        return String(format: NSLocalizedString("%d Days", comment: "Pump days remaining"), days)
    }
}

struct GlucoseGuardSparkline: View {
    let points: [Double]

    var body: some View {
        GeometryReader { geo in
            let values = points.isEmpty ? [90, 110, 95, 130, 100, 118, 102] : points
            let minV = min(values.min() ?? 0, 70) - 10
            let maxV = max(values.max() ?? 1, 180) + 10
            let span = max(maxV - minV, 1)
            Path { path in
                path.move(to: CGPoint(x: 0, y: geo.size.height * (1 - CGFloat((70 - minV) / span))))
                path.addLine(to: CGPoint(x: geo.size.width, y: geo.size.height * (1 - CGFloat((70 - minV) / span))))
            }
            .stroke(Color.white.opacity(0.12), style: StrokeStyle(lineWidth: 1, dash: [4, 4]))
            Path { path in
                path.move(to: CGPoint(x: 0, y: geo.size.height * (1 - CGFloat((180 - minV) / span))))
                path.addLine(to: CGPoint(x: geo.size.width, y: geo.size.height * (1 - CGFloat((180 - minV) / span))))
            }
            .stroke(GlucoseGuardTheme.green.opacity(0.28), style: StrokeStyle(lineWidth: 1, dash: [4, 4]))
            Path { path in
                for (index, value) in values.enumerated() {
                    let x = geo.size.width * CGFloat(index) / CGFloat(max(values.count - 1, 1))
                    let y = geo.size.height * (1 - CGFloat((value - minV) / span))
                    if index == 0 {
                        path.move(to: CGPoint(x: x, y: geo.size.height))
                        path.addLine(to: CGPoint(x: x, y: y))
                    } else {
                        path.addLine(to: CGPoint(x: x, y: y))
                    }
                }
                path.addLine(to: CGPoint(x: geo.size.width, y: geo.size.height))
                path.closeSubpath()
            }
            .fill(LinearGradient(colors: [GlucoseGuardTheme.red.opacity(0.38), Color.clear], startPoint: .top, endPoint: .bottom))
            Path { path in
                for (index, value) in values.enumerated() {
                    let x = geo.size.width * CGFloat(index) / CGFloat(max(values.count - 1, 1))
                    let y = geo.size.height * (1 - CGFloat((value - minV) / span))
                    if index == 0 {
                        path.move(to: CGPoint(x: x, y: y))
                    } else {
                        path.addLine(to: CGPoint(x: x, y: y))
                    }
                }
            }
            .stroke(GlucoseGuardTheme.red, lineWidth: 2.4)
            .shadow(color: GlucoseGuardTheme.red.opacity(0.55), radius: 8)
        }
    }
}
